
enter_communion_with_the_christians_pie = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp1_decisons_hold_bolt.dds"
	}
	decision_group_type = major

	desc = enter_communion_with_the_christians_pie_desc
	selection_tooltip = enter_communion_with_the_christians_pie_tooltip

	is_shown = {
		root.religion = religion:old_steppe
		NOT = {
			has_global_variable = pie_entered_communion_with_the_church
		}
	}

	is_valid = {
		custom_tooltip = {
			text = enter_communion_with_the_christians_requirement_tt
			religion:christianity_religion = {
				any_faith ={
					count >= 1
					main_rite = {
						head_of_rite = {
							opinion = {
								target = root
								value >= 50
							}
						}
					}
				}
			}
		}
		piety >= 1000
	}

	is_valid_showing_failures_only = {
		# Have to at _least_ not be in piety debt.
		piety >= 0
	}
    
	effect = {
		set_global_variable = {
			name = pie_entered_communion_with_the_church
			value = yes
		}
		pie_christianity_communion_scripted_effect = yes 
		trigger_event = { id = eme_religious_decision_events.1 }	
    }

    cost = {
		piety = 500
	}

	ai_check_interval = 96
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

synceticise_with_sinitic_world_pagan_rites_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp1_decisons_hold_bolt.dds"
	}
	decision_group_type = major

	desc = synceticise_with_sinitic_world_pagan_rites_decision_desc
	selection_tooltip = synceticise_with_sinitic_world_pagan_rites_decision_tooltip

	is_shown = {
		OR = {
			root.religion = {
				OR = {
					this = religion:kushitism_religion
					this = religion:old_steppe
					this = religion:retic_religion
				}
			}
			root.rite = {
				rite_has_doctrine = unreformed_faith_doctrine
			}
		}
		NOR = {
			root.rite = {
				has_variable = synceticised_with_sinitic_world
			}
		}
		root.location = {
			OR = {
				geographical_region = world_asia_china
				geographical_region = world_asia_japan
				geographical_region = world_asia_korea
			}
		}
	}

	is_valid = {
		OR = {
			AND = {
				custom_tooltip = {
					text = synceticise_with_sinitic_world_requirement_tt
					any_religion_global = {
						any_faith ={
							count >= 1
							main_rite.head_of_rite = {
								opinion = {
									target = root
									value >= 50
								}
								this.rite = {
									rite_has_doctrine = special_doctrine_immaterial_harmony
								}
							}
						}
					}
				}
				piety >= 800
			}
			piety >= 1600
		}	
	}

	is_valid_showing_failures_only = {
		# Have to at _least_ not be in piety debt.
		piety >= 0
	}
    
	effect = {
		root.rite = {
			set_variable = {
				name = synceticised_with_sinitic_world
				value = yes
			}
		}
		add_immaterial_harmony_to_pagan_rite_scripted_effect = yes 	
    }

    cost = {
		piety = 500
	}

	ai_check_interval = 96
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
