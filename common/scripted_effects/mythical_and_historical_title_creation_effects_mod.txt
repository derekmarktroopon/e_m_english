
restore_ithaca_scripted_effect = {
	dynasty = { add_dynasty_prestige = monumental_dynasty_prestige_gain }
	house = {
		add_house_modifier = {
			modifier = restorers_of_ithaca_house_modifier
		}
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}

	title:k_ithaca = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	if = {
		limit = {
			root.culture = {
				NOR = {
					has_name_list = name_list_ancient_greek
					has_name_list = name_list_mycenaean
				}
			}
		}
		root.culture = {
			save_temporary_scope_as = old_culture_to_get_innovations_from
		}
		set_culture = culture:ancient_greek
		culture:ancient_greek = {
			get_all_innovations_from = scope:old_culture_to_get_innovations_from
		}
		add_trait = eccentric
	} 
	title:c_cephalonia = {
		change_development_level = medium_development_level_gain
		set_county_culture = root.culture
		set_county_faith = root.faith
		set_county_rite = root.rite
		title_province = { 
			add_province_modifier = extra_building_slot
			add_province_modifier = restore_ithaca_modifier
		}
	}

	hidden_effect = {
		title:k_ithaca = {
			set_de_jure_liege_title = title:e_byzantium
		}
		title:d_cephalonia = {
			set_de_jure_liege_title = title:k_ithaca
		}
		title:d_kairouan = {
			set_de_jure_liege_title = title:k_carthage
		}
		title:c_buthrotum = {
			set_de_jure_liege_title = title:d_buthrotum
		}
		title:d_buthrotum = {
			set_de_jure_liege_title = title:k_epirus
		}
	}

	resolve_title_and_vassal_change = scope:title_change

	if = {
		limit = {
			capital_county != title:c_cephalonia
		}
		set_realm_capital = title:c_cephalonia
	}

	hidden_effect = {
		set_primary_title_to = title:k_ithaca
	}
}
