

tradition_hellenic_hoplites = {
	category = regional

	layers = {
		0 = martial
		1 = mena
		4 = battle.dds
	}

	is_shown = {
		OR = {
			this = culture:ancient_greek
			any_parent_culture_or_above = {
				this = culture:ancient_greek
			}
		}
		# DLC check.
		has_ep3_dlc_trigger = yes
	}
	can_pick = {
		culture_not_pacifistic_trigger = yes
	}
	
	parameters = {
		unlock_maa_hellenic_hoplites = yes
		blademaster_traits_more_common = yes
		prowess_from_martial_education = yes
		strong_traits_more_valued = yes
		strong_traits_more_common = yes
	}
	province_modifier = {
		levy_size = -0.25
	}
	character_modifier = {
		ai_boldness = 25
		monthly_martial_lifestyle_xp_gain_mult = 0.15
		men_at_arms_recruitment_cost = 0.25
		knight_effectiveness_mult = 1.5
		knight_limit = 2
		accolade_glory_gain_mult = 0.3
		men_at_arms_limit = -2
		men_at_arms_cap = -1
		pikemen_toughness_mult = 0.25
		pikemen_damage_mult = 0.1
		heavy_infantry_toughness_mult = 0.15
		heavy_infantry_damage_mult = 0.1
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_stoic = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_stoic_or_communal_desc
				}
			}
			if = {
				limit = {
					scope:character = {
						NOR = { # Culture Head need to have a 8 sized regiment of either Light/Heavy/Archer Cavalry
							number_maa_soldiers_of_base_type = {
								type = light_cavalry
								value >= 800
							}
							number_maa_soldiers_of_base_type = {
								type = heavy_cavalry
								value >= 400
							}
							number_maa_soldiers_of_base_type = {
								type = archer_cavalry
								value >= 800
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = culture_head_cavalry_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_hellenic_hoplites }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 200
		if = {
			limit = {
				scope:character ?= {
					ai_has_warlike_personality = yes
				}
			}
			multiply = 4
		}
	}
}
