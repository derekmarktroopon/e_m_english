tenet_bhakti = {
	icon = tenet_bhakti

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { rite_religion_tag = hinduism_religion }
				desc = tenet_bhakti_name
			}
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_venetic_deities_name
			}
			desc = tenet_bhakti_patron_god_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_venetic_deities_desc
			}
			desc = tenet_bhakti_desc
		}
	}

	piety_cost = faith_tenet_cost_mid

	is_shown = {
		OR = {
			religion_tag = hinduism_religion
			# With FP1, Germanics also get personal deities.
			AND = {
				religion_tag = germanic_religion
				has_fp1_dlc_trigger = yes
			}
			religion_tag = retic_religion
		}
	}

	can_pick = {
		OR = {
			religion = religion:hinduism_religion
			# With FP1, Germanics also get personal deities.
			trigger_if = {
				limit = { has_fp1_dlc_trigger = yes }
				religion = religion:germanic_religion
			}
			religion = religion:retic_religion
		}
		custom_tooltip = {
			text = incompatible_doctrine_monotheist
			NOT = { doctrine:doctrine_monotheist = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		select_personal_god_active
	}
}

tenet_megaliths = {
	icon = tenet_megaliths

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_camunic_arts_name
			}
			desc = tenet_megaliths_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_camunic_arts_desc
			}
			desc = tenet_megaliths_desc
		}
	}

	piety_cost = {
		if = {
			limit = {
				rite_has_tenet = tenet_natural_primitivism
			}
			value = faith_tenet_cost_low
		}
		else_if = {
			limit = { #Reduction in cost for Celtic Christianity because of stone circle shenanigans
				faith = faith:christian_faith
			}
			value = faith_tenet_cost_mid
		}
		else = {
			value = faith_tenet_cost_high
		}
		
		#We already practice megaliths
		if = {
			limit = { rite_has_tenet = tenet_megaliths }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		
	}

	parameters = {
		can_build_megaliths
		sanctify_artifacts_megalithic
	}

	character_modifier = {
		hills_development_growth_factor = 0.15
		hills_tax_mult = 0.1
		hills_levy_size = 0.3
		hills_advantage = 5
	}

	personal_tenet_modifier = {
		hills_construction_gold_cost = -0.1
		terraced_hills_construction_gold_cost = -0.1
		mountains_construction_gold_cost = -0.1
		desert_mountains_construction_gold_cost = -0.1
		build_speed = 0.25
	}

	personal_tenet_parameters = {
		tenet_megaliths_building_poi_double_rewards
		tenet_megaliths_rough_terrain_construction_fulfillment
	}

	traits = {
		# Megalithic builder cultures
		virtues = {
			architect = { weight = 15 scale = 1 }
			diligent = { weight = 10 scale = 1 }
		}
		sins = {
			lazy = { weight = 10 scale = 1 }
		}
	}
}

tenet_warmonger = {
	icon = tenet_warmonger

	name = {
		first_valid = {
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_religious_belligerence_name
			}
			desc = tenet_warmonger_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { religion = religion:retic_religion }
				desc = tenet_religious_belligerence_desc
			}
			desc = tenet_warmonger_desc
		}
	}

	piety_cost = {
		if = {
			limit = {
				OR = {
					rite_religion_tag = buddhism_religion
					rite_religion_tag = jainism_religion
					rite_religion_tag = taoism_religion
				}
			}
			value = faith_tenet_cost_massive
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_warmonger }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		NOR = {
			religion = religion:christianity_religion
			religion = religion:islam_religion
			religion = religion:judaism_religion
			religion = religion:dualism_religion
		}
		custom_description = {
			text = incompatible_tenet_pacifism_trigger
			NOT = { tenet:tenet_pacifism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = incompatible_tenet_dharmic_pacifism_trigger
			NOT = { tenet:tenet_dharmic_pacifism = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		warmonger
		conquest_cb_enabled
		invasion_cb_enabled
		great_holy_wars_active_if_reformed
		clergy_can_fight
		sanctify_artifacts_weapons
		faith_can_raid
	}

	special_parameters = {
		warmonger = yes
	}

	character_modifier = {
		ai_war_chance = 0.5
		ai_war_cooldown = -0.25
		accolade_glory_gain_mult = 0.1
	}

	personal_tenet_parameters = {
		clergy_can_fight_personal
	}

	personal_tenet_modifier = {
		prowess_per_piety_level = 1
		maa_damage_mult = 0.05
	}

	traits = {
		virtues = {
			brave = { weight = 20 scale = 1 }
			zealous = { weight = 10 scale = 1 }
		}
		sins = {
			craven = { weight = 20 scale = 1 }
		}
	}
}
