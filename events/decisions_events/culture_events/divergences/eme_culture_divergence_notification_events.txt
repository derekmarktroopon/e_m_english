##################################################
# Romano-Venetic becomes Venetian
##################################################

eme_culture_divergence_notification.1 = {  # Romano-Venetic becomes Venetian
	type = character_event
	title = eme_culture_divergence_notification.1.t  # Emergence of the Venetians
	desc = eme_culture_divergence_notification.1.desc # 
	theme = culture_change
	left_portrait = root

	# Option 1 (Almost always the only option): Accept the culture conversion.
	option = {
		name = eme_culture_divergence_notification.1.a
		set_culture = culture:venetian
		hidden_effect = {
			every_in_list = {
				list = romanoveneti_counties_to_make_venetian
				limit = {
					holder.top_liege = root.top_liege
				}
				set_county_culture = root.culture
			}
		}
	}

	# Option 2: I am strong enough to keep our old culture intact!
	option = {
		name = eme_culture_divergence_notification.1.b
		hidden_effect = {
			every_in_list = {
				list = romanoveneti_counties_to_make_venetian
				limit = {
					holder.top_liege = root.top_liege
				}
				set_county_culture = root.culture
			}
		}
	}
}

eme_culture_divergence_notification.2 = { # The changing ways of the Veneti
	type = character_event
	title = eme_culture_divergence_notification.2.t # Future of the Veneti
	desc = eme_culture_divergence_notification.2.desc # 
	theme = culture_change
	left_portrait = root

	# Option 1 (Almost always the only option): Accept the culture conversion.
	option = {
		name = eme_culture_divergence_notification.2.a # This is simply the way things are
	}

	# Option 2: I am strong enough to keep our old culture intact!
	option = {
		name = eme_culture_divergence_notification.2.b # No! We shall not forget our old ways!
		every_courtier = {
			limit = {
				culture = root.culture
			}
			set_culture = culture:venetic
		}
		set_culture = culture:venetic
		capital_county = { set_county_culture = culture:venetic }
	}
}
